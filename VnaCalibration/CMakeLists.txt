add_library(vna_calibration STATIC src/calibration.cpp)
target_include_directories(vna_calibration PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(vna_calibration PUBLIC cxx_std_14)
set_target_properties(vna_calibration PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(vna_calibration PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(BUILD_TESTING)
    add_executable(vna_calibration_tests tests/calibration_test.cpp)
    target_link_libraries(vna_calibration_tests PRIVATE vna_calibration)
    add_test(NAME calibration_math COMMAND vna_calibration_tests)
    set_tests_properties(calibration_math PROPERTIES TIMEOUT 10)
endif()
